-- CLOSEST PLAYER FUNCTION (Respect maxLockDistance and FOV)
local function getClosestPlayer()
	local mouse = localPlayer:GetMouse()
	local mousePos = Vector2.new(mouse.X, mouse.Y)
	local closestDistance = math.huge
	local selectedPlayer = nil

	for _, player in ipairs(Players:GetPlayers()) do
		if player ~= localPlayer and player.Character then
			local root = player.Character:FindFirstChild("HumanoidRootPart")
			local part = player.Character:FindFirstChild(targetPartName)

			if root and part then
				local distance = (localPlayer.Character.HumanoidRootPart.Position - root.Position).Magnitude
				if distance <= maxLockDistance then
					local screenPos, onScreen = camera:WorldToViewportPoint(part.Position)
					if onScreen then
						local screenDistance = (Vector2.new(screenPos.X, screenPos.Y) - mousePos).Magnitude
						-- Check if player is within FOV circle
						if screenDistance <= fovCircleRadius and screenDistance < closestDistance then
							closestDistance = screenDistance
							selectedPlayer = player
						end
					end
				end
			end
		end
	end

	return selectedPlayer
end

-- PREDICTION
local function getPredictedPosition(player)
	if not player or not player.Character then return nil end
	local part = player.Character:FindFirstChild(targetPartName)
	local root = player.Character:FindFirstChild("HumanoidRootPart")
	if not part or not root then return nil end

	local distance = (camera.CFrame.Position - root.Position).Magnitude
	local predictionTime = math.clamp(distance / 250, 0.01, 0.08)
	return part.Position + (root.Velocity * predictionTime)
end

-- ENHANCED CAMLOCK RENDER LOOP WITH HARD LOCK MODE
RunService.RenderStepped:Connect(function()
	if aimEnabled and camlockActive and lockedPlayer then
		local targetPos = getPredictedPosition(lockedPlayer)
		if targetPos then
			local currentCF = camera.CFrame
			local targetCF = CFrame.new(currentCF.Position, targetPos)
			
			-- Hard Lock Mode: 100% lock (instant) vs Smooth Lock
			if hardLockMode then
				camera.CFrame = targetCF -- Instant lock - perfect recoil control
			else
				camera.CFrame = currentCF:Lerp(targetCF, smoothingAmount) -- Smooth lock
			end
		end
	end
end)

-- INPUT HANDLING
UserInputService.InputBegan:Connect(function(input)
	if input.UserInputType == Enum.UserInputType.MouseButton2 and camlockActive then
		aimEnabled = true
		if not lockedPlayer then
			lockedPlayer = getClosestPlayer()
		end
	end

	if input.KeyCode == Enum.KeyCode.KeypadFive then
		guiVisible = not guiVisible
		if aimGuiRef then aimGuiRef.Enabled = guiVisible end
	end
end)

UserInputService.InputEnded:Connect(function(input)
	if input.UserInputType == Enum.UserInputType.MouseButton2 then
		aimEnabled = false
		lockedPlayer = nil
	end
end)

-- CHARACTER HANDLING (Fixed for :re command)
local function onCharacterAdded(character)
	-- Clean up old GUIs first
	cleanup()
	
	local humanoid = character:WaitForChild("Humanoid", 5)
	if humanoid then
		humanoid.Died:Connect(function()
			cleanup()
		end)
	end

	-- Wait a bit then recreate GUIs
	task.delay(2, function()
		if localPlayer.Character == character then -- Make sure character is still current
			createGUI()
			createFovCircle()
		end
	end)
end

-- Handle character removal (for :re command)
local function onCharacterRemoving(character)
	cleanup()
end

-- Connect character events
if localPlayer.Character then
	onCharacterAdded(localPlayer.Character)
end

localPlayer.CharacterAdded:Connect(onCharacterAdded)
localPlayer.CharacterRemoving:Connect(onCharacterRemoving)
